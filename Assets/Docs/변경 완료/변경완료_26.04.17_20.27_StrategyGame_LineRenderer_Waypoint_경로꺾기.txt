  using UnityEngine;

  public class BuildingLineConnector : MonoBehaviour
  {
      [SerializeField] private Spot _hqSpot;
      [SerializeField] private Spot[] _buildingSpots;
      [SerializeField] private LineRenderer[] _lineRenderers;
      [SerializeField] private float _lineZ = -1f;

      private void Start()
      {
          if (_buildingSpots.Length != _lineRenderers.Length)
          {
              Debug.LogError(
                  $"[BuildingLineConnector] buildingSpots({_buildingSpots.Length})와 " +
                  $"lineRenderers({_lineRenderers.Length})의 수가 다릅니다. Inspector를 확인하세요.");
              return;
          }

          Vector3 hqWorldPos = _hqSpot.transform.position;
          hqWorldPos.z = _lineZ;

          for (int i = 0; i < _buildingSpots.Length; i++)
          {
              Vector3 buildingWorldPos = _buildingSpots[i].transform.position;
              buildingWorldPos.z = _lineZ;

              _lineRenderers[i].useWorldSpace = true;
              _lineRenderers[i].positionCount = 2;
              _lineRenderers[i].SetPosition(0, hqWorldPos);
              _lineRenderers[i].SetPosition(1, buildingWorldPos);
          }
      }
  }

  using UnityEngine;

  [System.Serializable]
  public class RoadPath
  {
      public Spot buildingSpot;
      public LineRenderer lineRenderer;
      public Transform[] waypoints;
  }

  public class BuildingLineConnector : MonoBehaviour
  {
      [SerializeField] private Spot _hqSpot;
      [SerializeField] private RoadPath[] _roads;
      [SerializeField] private float _lineZ = -1f;

      private void Start()
      {
          Vector3 hqPos = _hqSpot.transform.position;
          hqPos.z = _lineZ;

          foreach (RoadPath road in _roads)
          {
              int count = 2 + road.waypoints.Length;
              road.lineRenderer.useWorldSpace = true;
              road.lineRenderer.positionCount = count;

              road.lineRenderer.SetPosition(0, hqPos);

              for (int i = 0; i < road.waypoints.Length; i++)
              {
                  Vector3 wp = road.waypoints[i].position;
                  wp.z = _lineZ;
                  road.lineRenderer.SetPosition(i + 1, wp);
              }

              Vector3 buildingPos = road.buildingSpot.transform.position;
              buildingPos.z = _lineZ;
              road.lineRenderer.SetPosition(count - 1, buildingPos);
          }
      }
  }

  [Play Mode 진입 시]

  1. Start() 호출
  2. HQ 월드 위치를 가져와 Z = _lineZ(-1)로 고정
  3. _roads 배열을 순서대로 순회 (foreach)
     각 RoadPath(road)에 대해:
       a. positionCount = 2 + waypoints 수 로 설정
          예) waypoint 2개 → positionCount = 4
       b. Position[0] = HQ 위치
       c. Position[1 ~ N-1] = waypoints 순서대로 위치
       d. Position[N] = Building 위치
       e. 모든 Z값 = _lineZ 로 강제 고정
  4. LineRenderer가 점들을 순서대로 연결하여 꺾인 경로를 그림

  [경로 예시 - waypoint 2개인 경우]
    Position[0] : HQ
    Position[1] : WP_01_01  ← 첫 번째 꺾임
    Position[2] : WP_01_02  ← 두 번째 꺾임
    Position[3] : Building

  [waypoint 0개인 경우]
    Position[0] : HQ
    Position[1] : Building
    → 기존과 동일한 직선

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  1. [Inspector 재연결 필수]
     _buildingSpots[], _lineRenderers[] 필드가 _roads[]로 바뀌어
     기존 Inspector 연결이 초기화된다.
     반드시 STEP 2에서 재연결해야 한다.

  2. [Waypoint 순서]
     Waypoints 배열 순서 = 경로 순서.
     HQ에서 Building 방향으로 순서대로 배열에 넣을 것.
     순서가 바뀌면 선이 교차하거나 꼬임.

  3. [Waypoint Z값]
     Waypoint 오브젝트의 Z는 코드에서 _lineZ로 자동 덮어씀.
     씬에서 Z를 신경 쓰지 않아도 됨.

  4. [Waypoints 없는 road]
     Waypoints Size = 0 으로 두면 기존처럼 직선으로 그려짐.
     road마다 waypoint 수를 다르게 해도 됨.

  5. [추후 캐릭터 이동 연동]
     캐릭터가 경로를 따라 이동할 때
     road.waypoints[] 배열을 그대로 이동 경로로 전달 가능.
     [HQ → waypoints → Building] 순서를 유지하면 됨.
