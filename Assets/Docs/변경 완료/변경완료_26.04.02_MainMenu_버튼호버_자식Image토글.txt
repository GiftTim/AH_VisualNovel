  -------------------------------------------------------
  using UnityEngine;
  using UnityEngine.UI;
  using UnityEngine.EventSystems;

  public class ButtonHoverEffect : MonoBehaviour,
      IPointerEnterHandler, IPointerExitHandler
  {
      private Image childImage;

      void Awake()
      {
          // 첫 번째 자식의 Image 컴포넌트를 자동으로 찾음
          childImage = transform.GetChild(0).GetComponent<Image>();
      }

      public void OnPointerEnter(PointerEventData eventData)
      {
          if (childImage != null)
              childImage.enabled = true;
      }

      public void OnPointerExit(PointerEventData eventData)
      {
          if (childImage != null)
              childImage.enabled = false;
      }
  }
  -------------------------------------------------------


----------------------------------------------------
4. 변경 후 작동 흐름
----------------------------------------------------

  [씬 시작]
    └─ 각 버튼 오브젝트의 ButtonHoverEffect.Awake() 실행
         └─ transform.GetChild(0).GetComponent<Image>()
              → 버튼의 첫 번째 자식 오브젝트에서 Image 컴포넌트를 찾아 childImage에 저장

  [마우스가 버튼 위에 올라감 (PointerEnter)]
    └─ Unity EventSystem이 OnPointerEnter() 호출
         └─ childImage.enabled = true
              → 자식 Image 컴포넌트 활성화 (화면에 표시됨)

  [마우스가 버튼 밖으로 나감 (PointerExit)]
    └─ Unity EventSystem이 OnPointerExit() 호출
         └─ childImage.enabled = false
              → 자식 Image 컴포넌트 비활성화 (화면에서 숨겨짐)
