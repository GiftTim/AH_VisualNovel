            missionUnit, missionUnit.Team, isSuccess, CurrentTime);

            _dayReport.HandleMissionSucceded(missionUnit.Gold);
        else
            _dayReport.HandleMissionFailed();
        missionUnit.ClaimMission();

    public void AcceptMission(MissionUnit mission, Team team)
    {
        _dayReport.HandleMissionAccepted(team.Members);

        mission.AcceptMission(team);
    }

    }

    }

    {
        _dayReport.AddCalls(newMissions.Count);
        OnMissionAvailable?.Invoke(newMissions);

3. 동작 흐름
================================================================

  씬 시작 시:
    DayManager.Start()
      → _missionManager.OnMissionMiss.AddListener(HandleMissionMiss)
      (임무 소멸 이벤트 구독 등록)

  StartDay(day, characters) 호출 시:
    1. 캐릭터 전원 Available 상태 초기화
    2. GameManager.Instance.GameState 참조
    3. DayReport 초기화
    4. daySO 선택 (날짜에 따라 또는 기본값)
    5. missionAmount 계산 (UseAllMissions이면 전체, 아니면 MissionAmount)
    6. DayCharacterManager.Init(characters)
    7. DayMissionManager.Init(missionSOs, missionAmount, dayDuration, HandleNewMissions)
       → 임무 셔플 → missionAmount 개 선택 → 등장 시간 고르게 배분
    8. DayLoopCoroutine() 시작

  DayLoopCoroutine() 실행 중 (0.5초마다):
    - DayCharacterManager.UpdateCharacters() 호출 (캐릭터 휴식 타이머 등)
    - DayMissionManager.UpdateMissions(elapsedTime) 호출
        → spawnTime 도달 임무: _pendingMissions → _activeMissions
        → HandleNewMissions 콜백 → DayReport.AddCalls() + OnMissionAvailable 발행
        → 각 activeMission.UpdateMission() 호출 (수락 타이머/완료 타이머 체크)
    - HasMissions()가 false가 되면 루프 종료
        → pendingMissions 비어있음 + activeMissions 전부 Claimed/Lost

  루프 종료 후:
    - WaitUntil(AllCharacterInBase) 대기
    - OnDayEnd(dayReport) 발행

  임무 수락 흐름 (외부에서 호출):
    AcceptMission(mission, team)
      → DayReport.HandleMissionAccepted()
      → DayCharacterManager.HandleTeamAcceptMission() (팀 → Going 상태)
      → MissionUnit.AcceptMission(team)

    StartMission(mission)
      → DayCharacterManager.HandleTeamStartMission() (팀 → InMission 상태)
      → MissionUnit.StartMission(CurrentTime) (완료 타이머 시작)

    ClaimMission(missionUnit, isSuccess)
      → DayCharacterManager.HandleTeamCompleteMission() (캐릭터 경험치 + 귀환 상태)
      → DayReport.HandleMissionSucceded/Failed()
      → MissionUnit.ClaimMission() (Claimed 상태로 전환)

4. 주의할 점
================================================================

  ① DayLoopCoroutine 파라미터 제거
     → 계획 h①에 따라 dayDuration 파라미터를 코루틴에서 제거했다.
     → DayDurationInSeconds는 _missionManager.Init() 호출 시 직접 전달된다.
     → DayLoopCoroutine()은 파라미터 없이 호출된다.

  ② HasMissions() 무한루프 방지
     → DayMissionManager.HasMissions()는 pendingMissions와 activeMissions를 모두 체크한다.
     → MissionUnit이 Lost 또는 Claimed 상태가 되면 카운트에서 제외된다.
     → 반드시 각 임무가 TimeToAccept 내에 수락되지 않으면 Lost가 되도록 MissionSO를 설정할 것.

  ③ PHASE 8-3, 8-4는 Unity Editor에서 수동 진행 필요
     → MissionSO 인스턴스(Mission_Test01.asset) 생성
     → Day_Default.asset에 MissionSOs 연결, DayDurationInSeconds=120, UseAllMissions=true
     → 씬 [Manager] 하위에 [DayMissionManager] 오브젝트 추가
     → DayMissionManager.cs 컴포넌트 부착
     → DayManager Inspector의 _missionManager 필드에 [DayMissionManager] 연결

  ④ StartDay() 호출 전 GameManager.NewGame() 선행 필요
     → _gameState = GameManager.Instance.GameState 참조가 있으므로
       GameManager.NewGame() 또는 LoadSave() 없이 StartDay()를 호출하면 NullReferenceException 발생.

  ⑤ MissionUnit.Gold 원본 버그 유지
     → MissionUnit.cs 197번째 줄: public int Gold => _missionSO.RewardExperience;
     → 원본 레포 버그 그대로 유지. 실제 골드 보상 필요 시 RewardGold로 수정.
