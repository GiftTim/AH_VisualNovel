    public Transform[] GetWaypoints(Spot targetSpot)
    {
        foreach (RoadPath road in _roads)
        {
            if (road.buildingSpot == targetSpot)
            {
                int count = 1 + road.waypoints.Length + 1;
                Transform[] path = new Transform[count];

                path[0] = _hqSpot.transform;

                for (int i = 0; i < road.waypoints.Length; i++)
                    path[i + 1] = road.waypoints[i];

                path[count - 1] = road.buildingSpot.transform;
                return path;
            }
        }
        return null;
    }

    using System;
    using System.Collections;
    using UnityEngine;
    using UnityEngine.Events;

    public class CharacterAgent : MonoBehaviour
    {
        [SerializeField] float _moveSpeed = 2f;

        public UnityEvent OnArrived;

        private Coroutine _moveCoroutine;

        public void MoveAlong(Transform[] path, Action onArrived = null)
        {
            if (_moveCoroutine != null) StopCoroutine(_moveCoroutine);
            _moveCoroutine = StartCoroutine(MoveCoroutine(path, onArrived));
        }

        IEnumerator MoveCoroutine(Transform[] path, Action onArrived)
        {
            if (path == null || path.Length < 2) yield break;

            for (int i = 0; i < path.Length - 1; i++)
            {
                Vector3 from = path[i].position;
                Vector3 to   = path[i + 1].position;
                float t = 0f;

                while (t < 1f)
                {
                    t += Time.deltaTime * _moveSpeed;
                    transform.position = Vector3.Lerp(from, to, Mathf.Clamp01(t));
                    yield return null;
                }
            }

            onArrived?.Invoke();
            OnArrived?.Invoke();
        }
    }

    using UnityEngine;

    namespace TESTING
    {
        public class TestingSpotState : MonoBehaviour
        {
            [SerializeField] Spot _targetSpot;
            [SerializeField] BuildingLineConnector _connector;
            [SerializeField] CharacterAgent _character;

            void Update()
            {
                if (Input.GetKeyDown(KeyCode.A))
                {
                    if (_targetSpot != null)
                    {
                        _targetSpot.SetState(BuildingState.Selected);

                        Transform[] path = _connector.GetWaypoints(_targetSpot);
                        if (path != null)
                            _character.MoveAlong(path);
                    }
                }

                if (Input.GetKeyDown(KeyCode.B))
                {
                    if (_targetSpot != null)
                        _targetSpot.SetState(BuildingState.Completed);
                }
            }
        }
    }

[주의할 점]
================================================================

  1. 캐릭터 초기 위치를 HQ와 동일하게 설정하지 않으면
     첫 이동 시 HQ 위치로 순간이동 후 이동 시작됨

  2. _targetSpot이 _connector에 등록되지 않은 경우 GetWaypoints()가 null 반환
     → TestingSpotState에서 null 체크 처리됨 (path != null 조건)

  3. 2D 게임이므로 waypoints의 z값 주의
     → BuildingLineConnector는 LineRenderer 표시를 위해 z = _lineZ(-1) 사용
     → 캐릭터가 waypoint z값을 그대로 따라가면 의도치 않은 z값이 될 수 있음
     → 필요 시 CharacterAgent에서 이동 후 z값을 고정값으로 덮어쓰는 처리 추가

  4. A 키 이동 중 재입력 시 기존 코루틴 중단 후 처음부터 재이동
     → 의도와 다를 경우 MoveAlong() 진입부에서 _moveCoroutine != null 시 return 처리 추가
