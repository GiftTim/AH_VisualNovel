    public bool dispaly_fullscreen = true;
    public string display_resolution = "1920 x 1080";

    public void Load()
    {
        var ui = ConfigMenu.instance.ui;

        // Apply the general settings
        ConfigMenu.instance.SetDisplayToFullScreen(dispaly_fullscreen);

        // Set the screen resolution
        int res_index = 0;
        for(int i = 0; i < ui.resolutions.options.Count; i++)
        {
            string resolution = ui.resolutions.options[i].text;
            if(resolution == display_resolution)
            {
                res_index = i;
                break;
            }
        }

        ui.resolutions.value = res_index;
    }

3. 변경 후 동작 흐름
----------------------------------------------------------------

  [게임 시작 시]
  1. ConfigMenu.Start() 호출
  2. SetAvailableResolutions() → 드롭다운에 고정 7개 항목 등록
       index 0: "1920 x 1080 (전체화면)"
       index 1~6: 창모드 해상도 목록
  3. LoadConfig() → vnconfig.cfg 존재 시 로드, 없으면 기본값 생성
  4. VN_Configuration.Load() 호출
       - display_resolution 문자열로 드롭다운 index 탐색
       - ui.resolutions.value = res_index (드롭다운 선택 반영)
       - ConfigMenu.instance.SetDisplayResolution() 직접 호출 → 실제 화면 적용

  [사용자가 드롭다운을 변경할 때 (OnValueChanged 연결 시)]
  1. SetDisplayResolution() 호출
  2. index == 0 → Screen.SetResolution(1920, 1080, FullScreenMode.ExclusiveFullScreen)
  3. index >= 1 → 텍스트 파싱 후 Screen.SetResolution(width, height, FullScreenMode.Windowed)
  4. confing.display_resolution = 선택된 텍스트 (저장용 갱신)

  [게임 종료 시]
  1. OnApplicationQuit() → VN_Configuration.Save()
  2. display_resolution 문자열이 JSON으로 vnconfig.cfg에 저장됨

  [저장 호환성]
  - 기존 cfg에 dispaly_fullscreen 필드가 있어도 JsonUtility가 무시 → 오류 없음
  - 기존 display_resolution = "1920 x 1080" → index 1과 일치, 창모드로 로드됨
  - 매칭 실패 시 res_index = 0 (전체화면)으로 fallback

1. 변경 이유
----------------------------------------------------------------

  기존 옵션 텍스트("1920 x 1080 (전체화면)", "1920 x 1080" 등)에서
  새 텍스트("전체창 (1920 x 1080)", "창모드 (1920 x 1080)" 등)로 변경 시
  파싱 로직 문제 발생:

    resolution.Split('x') → values[0] = "창모드 (1920 " → int.TryParse 실패

  해결 방법: 표시 텍스트와 저장값을 분리 (옵션 B)
    - 표시: "창모드 (1920 x 1080)"
    - 저장: "1920 x 1080"  (기존 파싱 가능한 형식 유지)

    public void SetDisplayResolution()
    {
        int index = ui.resolutions.value;
        string caption = ui.resolutions.captionText.text;
        string resolution = ExtractResolution(caption);   // "창모드 (1920 x 1080)" → "1920 x 1080"

        if (index == 0)
        {
            Screen.SetResolution(1920, 1080, FullScreenMode.ExclusiveFullScreen);
        }
        else
        {
            string[] values = resolution.Split('x');

            if(int.TryParse(values[0].Trim(), out int width) && int.TryParse(values[1].Trim(), out int height))
            {
                Screen.SetResolution(width, height, FullScreenMode.Windowed);
            }
            else
            {
                Debug.LogWarning($"Parsing error for screen resolution! ['{resolution}'] ...");
            }
        }

        confing.display_resolution = resolution;   // "1920 x 1080" 형태로 저장
    }

    // 새로 추가: "전체창 (1920 x 1080)" → "1920 x 1080", 괄호 없으면 원문 반환
    public string ExtractResolution(string caption)
    {
        int pStart = caption.IndexOf('(');
        int pEnd = caption.IndexOf(')');
        if (pStart >= 0 && pEnd > pStart)
            return caption.Substring(pStart + 1, pEnd - pStart - 1).Trim();
        return caption;
    }
