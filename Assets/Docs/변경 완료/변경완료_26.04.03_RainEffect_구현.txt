  using System.Collections;
  using UnityEngine;
  using UnityEngine.UI;

  public class RainEffect : MonoBehaviour
  {
      [SerializeField] Sprite dropletSprite;
      [SerializeField] Sprite rippleSprite;
      [SerializeField] Material dropletMaterial;
      [SerializeField] Material rippleMaterial;
      [SerializeField] float spawnInterval = 0.05f;
      [SerializeField] float fallSpeed = 800f;
      [SerializeField] int maxDroplets = 80;
      [SerializeField] Vector2 dropletSize = new Vector2(8, 40);
      [SerializeField] Vector2 rippleSize = new Vector2(60, 20);
      [SerializeField] float rippleDuration = 0.5f;
      [SerializeField] bool isPlaying = false;

      private RectTransform _rectTransform;
      private int _activeDroplets = 0;

      void Awake() { _rectTransform = GetComponent<RectTransform>(); }
      void Start() { if (isPlaying) Play(); }

      public void Play()  { isPlaying = true; StartCoroutine(SpawnLoop()); }
      public void Stop()  { isPlaying = false; }

      IEnumerator SpawnLoop()       // 일정 간격으로 Droplet 스폰
      IEnumerator DropletFall()     // 낙하 이동 + 하단 도달 시 Ripple 호출
      IEnumerator RippleAnimation() // Scale 확대 + Alpha Fade Out
  }
  ----------------------------------------------------------

================================================================
  3. 동작 흐름
================================================================

  [시작 조건]
    - Play()를 외부에서 호출하거나, 인스펙터에서 isPlaying = true 로 설정 후 실행

  [SpawnLoop 코루틴]
    1. isPlaying == true 인 동안 spawnInterval(0.05초) 간격으로 반복
    2. 현재 활성 빗방울 수(_activeDroplets)가 maxDroplets(80) 미만이면
       SpawnDroplet() 호출

  [SpawnDroplet]
    1. "Droplet" GameObject를 RainEffect 오브젝트 하위에 생성
    2. RectTransform: 크기 = dropletSize(8×40)
                      X = 패널 너비 내 랜덤, Y = 패널 상단 위
    3. Image 컴포넌트: dropletSprite + Additive_Droplet 머티리얼 적용
    4. _activeDroplets++ 후 DropletFall 코루틴 시작

  [DropletFall 코루틴]
    1. 매 프레임 anchoredPosition.y를 fallSpeed(800px/s) 속도로 감소
    2. 패널 하단(bottomY)을 벗어나면:
       - Droplet GameObject 제거
       - _activeDroplets--
       - 착지 위치에서 RippleAnimation 코루틴 시작

  [RippleAnimation 코루틴]
    1. 착지 위치에 "Ripple" GameObject 생성
    2. Image 컴포넌트: rippleSprite + Additive_Ripple 머티리얼 적용
    3. rippleDuration(0.5초) 동안:
       - sizeDelta : (0,0) → rippleSize(60×20) 으로 선형 확대
       - color.a   : 1.0 → 0.0 으로 Fade Out
    4. 애니메이션 완료 후 Ripple GameObject 제거

  [종료]
    - Stop() 호출 시 isPlaying = false → SpawnLoop 코루틴 자연 종료
    - 이미 낙하 중인 Droplet은 패널 하단 도달까지 계속 동작 후 제거
