[파일: CharacterAgent.cs] ← 신규 생성
  경로: Assets/Scripts/StrategyGame/Core/CharacterAgent.cs

  using System;
  using System.Collections;
  using UnityEngine;
  using UnityEngine.Events;

  public class CharacterAgent : MonoBehaviour
  {
      [SerializeField] float _moveSpeed = 2f;

      public UnityEvent OnArrived;

      private Coroutine _moveCoroutine;

      public void MoveAlong(Transform[] path, Action onArrived = null)
      {
          if (_moveCoroutine != null) StopCoroutine(_moveCoroutine);
          _moveCoroutine = StartCoroutine(MoveCoroutine(path, onArrived));
      }

      IEnumerator MoveCoroutine(Transform[] path, Action onArrived)
      {
          if (path == null || path.Length < 2) yield break;

          for (int i = 0; i < path.Length - 1; i++)
          {
              Vector3 from = path[i].position;
              Vector3 to   = path[i + 1].position;
              float t = 0f;

              while (t < 1f)
              {
                  t += Time.deltaTime * _moveSpeed;
                  transform.position = Vector3.Lerp(from, to, Mathf.Clamp01(t));
                  yield return null;
              }
          }

          onArrived?.Invoke();
          OnArrived?.Invoke();
      }
  }

7. 주의할 점 (h)
================================================================

  1. 캐릭터 초기 위치
     - CharacterAgent 오브젝트의 초기 위치를 HQ의 위치와 동일하게 설정해야 함.
     - GetWaypoints()가 반환하는 path[0]이 HQ이므로
       캐릭터가 HQ에서 시작하지 않으면 첫 이동 시 순간이동처럼 보임.

  2. GetWaypoints()가 null을 반환하는 경우
     - _targetSpot이 BuildingLineConnector에 등록되지 않은 경우 null 반환.
     - TestingSpotState에서 null 체크 필요 (이미 위 코드에 반영됨).
     - 추가로 Debug.LogWarning으로 알림 삽입 권장.

  3. Waypoint의 z값 처리
     - BuildingLineConnector에서 LineRenderer 표시를 위해 z = _lineZ(-1)로 설정.
     - CharacterAgent 이동 시 waypoints의 실제 z값은 Transform의 world position.
     - 2D 게임이므로 캐릭터의 z값을 별도로 고정하거나
       CharacterAgent에서 이동 후 z를 원하는 값으로 덮어쓰는 처리 필요.
       예: transform.position = new Vector3(pos.x, pos.y, 원하는z값);

  4. 이동 중 재이동 요청
     - A 키를 이동 중에 다시 누르면 기존 코루틴이 중단되고 처음부터 재이동.
     - 이 동작이 의도와 맞는지 확인. 무시하려면 _moveCoroutine != null 시
       return 처리 추가.

  5. DOTween과의 충돌
     - 프로젝트에 DOTween이 포함되어 있음.
     - 같은 Transform.position을 Lerp 코루틴과 DOTween이 동시에 조작하면 충돌.
     - 캐릭터 오브젝트에는 한 가지 이동 방식만 사용.

  6. 나중에 Mission 시스템으로 이관 시
     - 현재 TestingSpotState에서 직접 처리하는 로직
       (SetState + GetWaypoints + MoveAlong)을
       MissionManager 또는 QuestCoordinator로 이전하면 됨.
     - TestingSpotState는 테스트용이므로 이 파일을 그대로 참고 후 제거.

  7. BuildingLineConnector의 [ExecuteAlways] 속성
     - 현재 BuildingLineConnector는 [ExecuteAlways]로 에디터에서도 Update 실행.
     - GetWaypoints()는 단순 조회 메서드이므로 영향 없음.
