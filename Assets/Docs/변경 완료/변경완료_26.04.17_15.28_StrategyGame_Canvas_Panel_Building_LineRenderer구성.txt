  ---------------------------------------------------------
  using UnityEngine;

  public class BuildingLineConnector : MonoBehaviour
  {
      [SerializeField] private Spot _hqSpot;
      [SerializeField] private Spot[] _buildingSpots;
      [SerializeField] private LineRenderer[] _lineRenderers;

      private void Start()
      {
          if (_buildingSpots.Length != _lineRenderers.Length)
          {
              Debug.LogError(
                  $"[BuildingLineConnector] buildingSpots({_buildingSpots.Length})와 " +
                  $"lineRenderers({_lineRenderers.Length})의 수가 다릅니다. Inspector를 확인하세요.");
              return;
          }

          Vector3 hqWorldPos = _hqSpot.transform.position;

          for (int i = 0; i < _buildingSpots.Length; i++)
          {
              Vector3 buildingWorldPos = _buildingSpots[i].transform.position;

              _lineRenderers[i].useWorldSpace = true;
              _lineRenderers[i].positionCount = 2;
              _lineRenderers[i].SetPosition(0, hqWorldPos);
              _lineRenderers[i].SetPosition(1, buildingWorldPos);
          }
      }
  }
  ---------------------------------------------------------

  [Play Mode 진입 시]

  1. BuildingLineConnector.Start() 호출
  2. _buildingSpots.Length != _lineRenderers.Length 이면
     → Debug.LogError 출력 후 처리 중단
  3. _hqSpot.transform.position 으로 HQ 월드 좌표 취득
  4. i = 0 ~ (_buildingSpots.Length - 1) 반복 :
       a. _buildingSpots[i].transform.position 으로 해당 건물 월드 좌표 취득
       b. _lineRenderers[i].useWorldSpace = true  설정
       c. _lineRenderers[i].positionCount = 2     설정
       d. SetPosition(0, hqWorldPos)    → 선의 시작점 = HQ 중심
       e. SetPosition(1, buildingWorldPos) → 선의 끝점 = Building 중심
  5. 결과 : 각 Building마다 HQ 중심점으로부터 선이 그어짐

  [좌표계 흐름]
  RectTransform.position (Screen Space-Camera Canvas의 UI 오브젝트)
  → Unity 내부에서 월드 좌표로 자동 변환되어 있음
  → LineRenderer(useWorldSpace=true)와 동일 좌표계 사용 가능

  5. 주의할 점
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  1. [좌표계]
     transform.position 은 Screen Space-Camera Canvas 아래 UI 오브젝트의
     경우 월드 좌표로 반환된다.
     RectTransform.anchoredPosition 을 사용하면 캔버스 로컬 좌표가 되어
     LineRenderer와 좌표계가 달라지므로 사용하지 말 것.

  2. [배열 길이 불일치]
     _buildingSpots 와 _lineRenderers 의 Element 수가 다르면
     Start() 에서 에러 로그를 출력하고 선이 그려지지 않는다.
     Building을 추가/삭제할 때 두 배열을 항상 동시에 수정할 것.

  3. [LineRenderer Sorting]
     LineRenderer 는 기본 Default Sorting Layer 에 배치된다.
     Canvas의 Order in Layer 보다 높게 설정하면 UI 위에 그려진다.
     패널 순서에 맞춰 Sorting Layer / Order를 조정할 것.

  4. [Canvas Render Camera 미할당]
     Screen Space - Camera 모드에서 카메라를 할당하지 않으면
     Overlay처럼 동작하여 LineRenderer와 렌더 순서가 어긋난다.
     반드시 카메라를 할당할 것.

  5. [Spot.cs 컴파일 후 확인]
     using 변경 후 Unity가 재컴파일할 때 기존에 Microsoft.Unity 타입으로
     직렬화된 _icon 필드 값이 누락될 수 있다.
     SpotVisualConfig._icon 이 Inspector에서 사라지면 재할당 필요.
